Get balance

Returns the number of credits left on your account. Free.

GET/v1/balance

Request

curl {BASE}/v1/balance \
  -H "Authorization: Bearer $KAALERA_API_KEY"
const res = await fetch("{BASE}/v1/balance", {
  headers: { Authorization: `Bearer ${process.env.KAALERA_API_KEY}` },
});
const data = await res.json();
console.log(res.status, data);
import os, requests

res = requests.get(
    "{BASE}/v1/balance",
    headers={"Authorization": f"Bearer {os.environ['KAALERA_API_KEY']}"},
)
print(res.status_code, res.json())

Response

{
  "credits": 3
}
